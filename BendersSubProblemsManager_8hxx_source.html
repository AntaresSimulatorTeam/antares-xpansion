161 void GetSubproblemCut(SubProblemDataMap& subproblem_data_map,
162 const FastBeginHook& fast_begin_hook,
163 const CacheBeginHook& cache_begin_hook,
164 const PostSolveHook& post_solve_hook)
165 {
166 switch (options_.CACHE_PROBLEMS)
167 {
168 case CacheProblems::NO_CACHE:
169 GetSubproblemCutFast(subproblem_data_map, fast_begin_hook, post_solve_hook);
170 break;
171 case CacheProblems::PER_SUB:
172 GetSubproblemCutCache(subproblem_data_map, cache_begin_hook, post_solve_hook);
173 break;
174 case CacheProblems::COMPACT:
175 GetCompactInMemCuts(subproblem_data_map, cache_begin_hook, post_solve_hook);
176 break;
177 default:
178 break;
179 }
180 }

182 // ---------------------------------------------------------------
183 // Execution policy selection
184 // ---------------------------------------------------------------
185
186 // Subproblems are solved with Derived::EXECUTION_POLICY, except when micro-iterations are
187 // enabled: the plugin hooks then mutate state shared between subproblems (constraint maps,
188 // the shared skeleton solver) and call into a dynamically loaded library, none of which is
189 // guaranteed to be thread-safe. See BendersPlugin.
190 template<typename It, typename F>
191 void ForEachSubproblem(It first, It last, F&& f)
192 {
193 if (options_.MICRO_ITERATIONS)
194 {
195 std::for_each(std::execution::seq, first, last, std::forward<F>(f));
196 }
197 else
198 {
199 std::for_each(Derived::EXECUTION_POLICY, first, last, std::forward<F>(f));
200 }
201 }

203 // ---------------------------------------------------------------
204 // Cache=0: fast path — persistent workers
205 // ---------------------------------------------------------------
206
207 void GetSubproblemCutFast(SubProblemDataMap& subproblem_data_map,
208 const FastBeginHook& begin_hook,
209 const PostSolveHook& post_solve_hook)
210 {
211 auto nameAndWorkers = begin_hook();
212
213 std::mutex m;
214 std::exception_ptr first_exception;
215 ForEachSubproblem(nameAndWorkers.begin(),
216 nameAndWorkers.end(),
217 [this, &m, &subproblem_data_map, &first_exception, &post_solve_hook](
218 const std::pair<std::string, SubproblemWorkerPtr>& kvp)
219 {
220 try
221 {
222 PlainData::SubProblemData subproblem_data;
223 const auto& [name, worker] = kvp;
224 SolveSubproblem(subproblem_data, name, worker, nullptr);
225 post_solve_hook(name, subproblem_data, worker);
226
227 std::lock_guard guard(m);
228 subproblem_data_map[name] = subproblem_data;
229 }
230 catch (...)
231 {
232 std::lock_guard guard(m);
233 if (!first_exception)
234 {
235 first_exception = std::current_exception();
236 }
237 }
238 });
239 if (first_exception)
240 {
241 std::rethrow_exception(first_exception);
242 }
243 }

245 // ---------------------------------------------------------------
246 // Cache=1: disk cache — recreate workers each iteration
247 // ---------------------------------------------------------------
248
249 void GetSubproblemCutCache(SubProblemDataMap& subproblem_data_map,
250 const CacheBeginHook& begin_hook,
251 const PostSolveHook& post_solve_hook)
252 {
253 auto nameAndVariableMap = begin_hook();
254
255 std::mutex m;
256 std::exception_ptr first_exception;
257
258 ForEachSubproblem(nameAndVariableMap.begin(),
259 nameAndVariableMap.end(),
260 [this, &m, &subproblem_data_map, &first_exception, &post_solve_hook](
261 const std::pair<std::string, VariableMap>& kvp)
262 {
263 try
264 {
265 const auto& [name, variables] = kvp;
266 auto worker = makeSubproblemWorker(kvp);
267 PlainData::SubProblemData subproblem_data;
268 SolveSubproblem(subproblem_data,
269 name,
270 worker,
271 [this, &name, &worker]
272 { TryRestoreSubproblemBasis(name, worker); });
273 post_solve_hook(name, subproblem_data, worker);
274
275 std::call_once(
276 variable_indice_once_flag_,
277 [this](const auto& worker_)
278 { SetSubproblemVariablesIndices(worker_); },
279 *worker);
280
281 StoreSubproblemBasis(name, worker);
282
283 std::lock_guard guard(m);
284 subproblem_data_map[name] = subproblem_data;
285 }
286 catch (...)
287 {
288 std::lock_guard guard(m);
289 if (!first_exception)
290 {
291 first_exception = std::current_exception();
292 }
293 }
294 });
295 if (first_exception)
296 {
297 std::rethrow_exception(first_exception);
298 }
299 }

301 // ---------------------------------------------------------------
302 // Cache=2: skeleton — shared solver, morphed per subproblem
303 // ---------------------------------------------------------------
304
305 void GetCompactInMemCuts(SubProblemDataMap& subproblem_data_map,
306 const CacheBeginHook& begin_hook,
307 const PostSolveHook& post_solve_hook)
308 {
309 auto nameAndVariableMap = begin_hook();
310
311 for (auto& [sub, variables]: nameAndVariableMap)
312 {
313 double slave_weights = SubproblemWeight(data_.control.nsubproblem, sub);
314
315 auto subproblem_worker = subproblem_worker_factory_
316 ->CreateSubSolverAbstract(sub, variables, slave_weights);
317
318 PlainData::SubProblemData subproblem_data;
319 SolveSubproblem(subproblem_data,
320 sub,
321 subproblem_worker,
322 [this, &sub] { subproblem_worker_factory_->ApplyBasis(sub); });
323 post_solve_hook(sub, subproblem_data, subproblem_worker);
324
325 subproblem_worker_factory_->GetBasis(sub);
326
327 subproblem_data_map[sub] = subproblem_data;
328 }
329 }

331 // ---------------------------------------------------------------
332 // Core solve (uniform for all variants — no virtual override)
333 // ---------------------------------------------------------------
334
335 void SolveSubproblem(PlainData::SubProblemData& subproblem_data,
336 const std::string& name,
337 const std::shared_ptr<SubproblemWorker>& worker,
338 const std::function<void()>& post_reset_hook)
339 {
340 Timer subproblem_timer;
341 worker->fix_to(data_.solution.x_cut);
342 plugin_->OnBendersSubResolutionStart(worker, name);
343 // with this hook we try to avoid duplicating the whole bloc since the diffence is just the
344 // overhead of the method finally so for every special case we just set a lambda function
345 if (post_reset_hook && plugin_->ShouldRestoreSubproblemBasis())
346 {
347 post_reset_hook();
348 }
349
350 int num_micro_iter(0);
351 if (options_.MICRO_ITERATIONS)
352 {
353 bool added_rows = true;
354 plugin_->OnBendersMicroIterationStart();
355 while (added_rows)
356 {
357 auto t1 = std::chrono::steady_clock::now();
358 worker->solve(subproblem_data.lpstatus,
359 options_.OUTPUTROOT,
360 options_.LAST_MASTER_MPS + MPS_SUFFIX,
361 writer_);
362
363 auto t2 = std::chrono::steady_clock::now();
364 auto elapsed_microseconds = std::chrono::duration_cast<std::chrono::microseconds>(
365 t2 - t1)
366 .count();
367
368 num_micro_iter++;
369 plugin_->OnBendersMicroIterationEnd(name,
370 added_rows,
371 std::to_string(elapsed_microseconds),
372 data_.control.it,
373 num_micro_iter);
374 }
375 }
376 else
377 {
378 worker->solve(subproblem_data.lpstatus,
379 options_.OUTPUTROOT,
380 options_.LAST_MASTER_MPS + MPS_SUFFIX,
381 writer_);
382 }
383
384 worker->get_value(subproblem_data.subproblem_cost);
385 worker->get_subgradient(subproblem_data.var_name_and_subgradient);
386 worker->get_splex_num_of_ite_last(subproblem_data.simplex_iter);
387 subproblem_data.subproblem_timer = subproblem_timer.elapsed();
388
389 plugin_->OnBendersSubResolutionEnd();
390 }
