17class SolverLogManager
18{
19public:
20 explicit SolverLogManager() = default;

22 explicit SolverLogManager(const SolverLogManager& other):

27 explicit SolverLogManager(const std::filesystem::path& log_file):
28 log_file_path(log_file)
29 {
30 init();
31 }
32
33 SolverLogManager& operator=(const SolverLogManager& other)
34 {
35 if (this == &other)
36 {
37 return *this;
38 }
39 log_file_path = other.log_file_path;
40 init();
41 return *this;
42 }
43
44 void init()
45 {
46#ifdef __linux__
47 if (log_file_path.empty()
48 || (log_file_ptr = fopen(log_file_path.string().c_str(), "a+")) == nullptr)
49#elif _WIN32
50 if (log_file_path.empty()
51 || (log_file_ptr = _fsopen(log_file_path.string().c_str(), "a+", _SH_DENYNO))
52 == nullptr)
53#endif
54 {
55 std::cout << "Invalid log file name passed as parameter: "
56 << std::quoted(log_file_path.string()) << std::endl;
57 }
58 else
59 {
60 setvbuf(log_file_ptr, nullptr, _IONBF, 0);
61 }
62 }
63
64 ~SolverLogManager()
65 {
66 if (log_file_ptr)
67 {
68 fclose(log_file_ptr);
69 log_file_ptr = nullptr;
70 }
71 }
72
73 FILE* log_file_ptr = nullptr;
74 std::filesystem::path log_file_path = "";
