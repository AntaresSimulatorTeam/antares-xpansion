383 int get_nrows() const override

385 return nrows_;

395 void set_obj(const double* obj, int first, int last) override

407 constraints_coeffs_ = constraints;

410 void chg_obj(const std::vector<int>& mindex, const std::vector<double>& obj) override
411 {
412 for (size_t i = 0; i < obj.size(); ++i)
413 {
414 obj_coeffs_[mindex[i]] = obj[i];
415 }

418 void chg_rhs_values(std::vector<int>& id_rows, std::vector<double>& vals) override
419 {

426 void chg_coefs(const std::vector<int>& id_rows,
427 const std::vector<int>& id_cols,
428 const std::vector<double>& vals) override
429 {
430 for (size_t i = 0; i < vals.size(); ++i)
431 {
432 constraints_coeffs_[id_rows[i]][id_cols[i]] = vals[i];
433 }
434 }

436 void get_obj(double* obj, int first, int last) const override
437 {
438 std::copy(obj_coeffs_.begin() + first, obj_coeffs_.begin() + last + 1, obj);
439 }

442 {

444 }
445
446 int get_col_index(const std::string& name) override
447 {
448 auto it = col_name_index_map_.find(name);
449 return it == col_name_index_map_.end() ? -1 : it->second;
450 }
451
452 int get_row_index(const std::string& name) override
453 {
454 auto it = row_name_index_map_.find(name);
455 return it == row_name_index_map_.end() ? -1 : it->second;
456 }
457
458 void get_rows(int* mstart,
459 int* mclind,
460 double* dmatval,
461 int /*size*/,
462 int* nels,
463 int first,
464 int last) const override
465 {
466 int nz = 0;
467 for (int r = first; r <= last; ++r)
468 {
469 mstart[r - first] = nz;
470 const auto& row = constraints_coeffs_[r];
471 for (size_t c = 0; c < row.size(); ++c)
472 {
473 mclind[nz] = static_cast<int>(c);
474 dmatval[nz] = row[c];
475 ++nz;
476 }
477 nels[r - first] = static_cast<int>(row.size());
478 }
479 }
480
481 int ncols_ = 0;
482 int nrows_ = 0;
483 std::vector<double> obj_coeffs_;
484 std::vector<double> rhs_coeffs_;
485 std::vector<std::vector<double>> constraints_coeffs_;
486 std::map<std::string,int> col_name_index_map_ ;
487 std::map<std::string,int> row_name_index_map_ ;
488};
