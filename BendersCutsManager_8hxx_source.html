30 // Aggregates subproblem costs and tracks min/max simplex iterations
31 // across all gathered subproblem data. Called during cut building to
32 // populate iteration-level statistics before constructing the cuts.
33 void SetSubproblemDataCostAndSimplexIter(
34 const std::vector<SubProblemDataMap>& gathered_subproblem_map,
35 CurrentIterationData& data)
36 {
37 for (const auto& subproblem_data_map: gathered_subproblem_map)
38 {
39 for (auto&& [sub_problem_name, subproblem_data]: subproblem_data_map)
40 {
41 data.cuts.subproblem_cost += subproblem_data.subproblem_cost;
42 BoundSimplexIterations(subproblem_data.simplex_iter, data);
43 }
44 }
45 }

47 // Computes the separation point x_cut from x_out and x_in using the
48 // separation parameter, then rounds values near variable bounds.
49 void ComputeXCut(CurrentIterationData& data,
50 double separation_param,
51 double master_solution_tolerance)
52 {
53 if (data.control.it == 1)
54 {
55 data.solution.x_in = data.solution.x_out;
56 data.solution.x_cut = data.solution.x_out;
57 data.solution.master_only_vars_in = data.solution.master_only_vars_out;
58 data.solution.master_only_vars_cut = data.solution.master_only_vars_out;
59 }
60 else
61 {
62 for (const auto& [name, value]: data.solution.x_out)
63 {
64 data.solution.x_cut[name] = separation_param * data.solution.x_out[name]
65 + (1 - separation_param) * data.solution.x_in[name];
66 }
67 for (int i(0); i < data.solution.master_only_vars_out.size(); ++i)
68 {
69 data.solution.master_only_vars_cut[i] = separation_param
70 * data.solution.master_only_vars_out[i]
71 + (1 - separation_param)
72 * data.solution.master_only_vars_in[i];
73 }
74 }
75 RoundXCut(data, master_solution_tolerance);
76 }

78 void BuildAllAggregatedCuts(const std::vector<SubProblemNamesInCut>& subproblem_names,
79 const std::vector<SubProblemDataMap>& gathered_subproblem_map,
80 const VariableMap& problem_to_id,
81 double& ub,
82 const Point& x_cut,
83 SubProblemDataMap& cut_trace)
84 {
85 assert((subproblem_names.empty() || master_manager_.Master() != nullptr)
86 && "cuts can only be built on the process owning the master problem");
87
88 for (const auto& subproblem_names_in_cut: subproblem_names)
89 {
90 Point s;
91 double rhs{0};
92 std::vector<int> subproblem_ids_per_cut;
93
94 for (const auto& [sub_problem_name, position_in_gathered]: subproblem_names_in_cut)
95 {
96 // A subproblem of the cut with no id or no gathered data would silently yield
97 // a cut that does not bound the subproblem (missing rhs / subgradient), and so
98 // an invalid lower approximation: fail instead.
99 const auto problem_to_id_pair = problem_to_id.find(sub_problem_name);
100 if (problem_to_id_pair == problem_to_id.end())
101 {
102 throw LogUtils::XpansionError<std::runtime_error>(
103 "Cannot build cut: subproblem " + sub_problem_name + " has no id",
104 LOGLOCATION);
105 }
106 if (position_in_gathered < 0
107 || static_cast<size_t>(position_in_gathered) >= gathered_subproblem_map.size())
108 {
109 throw LogUtils::XpansionError<std::runtime_error>(
110 "Cannot build cut: subproblem " + sub_problem_name
111 + " refers to gathered position " + std::to_string(position_in_gathered)
112 + " but only " + std::to_string(gathered_subproblem_map.size())
113 + " were gathered",
114 LOGLOCATION);
115 }
116 const auto& gathered_for_position = gathered_subproblem_map[position_in_gathered];
117 const auto subproblem_data_pair = gathered_for_position.find(sub_problem_name);
118 if (subproblem_data_pair == gathered_for_position.end())
119 {
120 throw LogUtils::XpansionError<std::runtime_error>(
121 "Cannot build cut: no data gathered for subproblem " + sub_problem_name
122 + " at position " + std::to_string(position_in_gathered),
123 LOGLOCATION);
124 }
125
126 subproblem_ids_per_cut.push_back(problem_to_id_pair->second);
127 const auto& subproblem_data = subproblem_data_pair->second;
128 ub += subproblem_data.subproblem_cost;
129 rhs += subproblem_data.subproblem_cost;
130 compute_cut_val(subproblem_data.var_name_and_subgradient, x_cut, s);
131 cut_trace[sub_problem_name] = subproblem_data;
132 }
133
134 master_manager_.AddGroupSubproblemCut(subproblem_ids_per_cut, s, x_cut, rhs);
135 }
136 }

142 // Accumulates the subgradient of one subproblem into the cut coefficients s,
143 // for every candidate present in the separation point x_cut.
144 static void compute_cut_val(const Point& var_name_subgradient, const Point& x_cut, Point& s)
145 {
146 for (const auto& [cand_name, cand_value]: x_cut)
147 {
148 const auto cand_name_and_subgradient = var_name_subgradient.find(cand_name);
149 if (cand_name_and_subgradient != var_name_subgradient.end())
150 {
151 s[cand_name] += cand_name_and_subgradient->second;
152 }
153 }
154 }

156 // Rounds x_cut values that are within tolerance of variable bounds to
157 // avoid numerical drift from repeated separation parameter application.
158 void RoundXCut(CurrentIterationData& data, double master_solution_tolerance)
159 {
160 for (auto& kvp: data.solution.x_cut)
161 {
162 double value = kvp.second;
163 double lb = data.solution.min_invest.at(kvp.first);
164 double ub = data.solution.max_invest.at(kvp.first);
165
166 if (std::abs(value - lb) < master_solution_tolerance)
167 {
168 kvp.second = lb;
169 }
170 else if (std::abs(value - ub) < master_solution_tolerance)
171 {
172 kvp.second = ub;
173 }
174 }
175 }

177 // Updates min/max simplex iteration bounds for the current iteration.
178 // Used by SetSubproblemDataCostAndSimplexIter to track solver effort
179 // across subproblems during cut gathering.
180 void BoundSimplexIterations(int subproblem_iterations, CurrentIterationData& data)
181 {
182 data.cuts.max_simplexiter = (data.cuts.max_simplexiter < subproblem_iterations)
183 ? subproblem_iterations
184 : data.cuts.max_simplexiter;
185 data.cuts.min_simplexiter = (data.cuts.min_simplexiter > subproblem_iterations)
186 ? subproblem_iterations
187 : data.cuts.min_simplexiter;
188 }
